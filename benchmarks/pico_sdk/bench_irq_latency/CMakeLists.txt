cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(bench_irq_latency C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(bench_irq_latency main.c)
target_compile_options(bench_irq_latency PRIVATE -O3 -flto)
target_link_options(bench_irq_latency PRIVATE -flto)
target_link_libraries(bench_irq_latency pico_stdlib hardware_timer hardware_irq hardware_clocks)
pico_enable_stdio_uart(bench_irq_latency 1)
pico_enable_stdio_usb(bench_irq_latency 0)
pico_add_extra_outputs(bench_irq_latency)
