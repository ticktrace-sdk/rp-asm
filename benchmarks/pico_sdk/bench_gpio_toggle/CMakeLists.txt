cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(bench_gpio_toggle C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_executable(bench_gpio_toggle main.c)
target_compile_options(bench_gpio_toggle PRIVATE -O3 -flto)
target_link_options(bench_gpio_toggle PRIVATE -flto)
target_link_libraries(bench_gpio_toggle pico_stdlib hardware_clocks)
pico_enable_stdio_uart(bench_gpio_toggle 1)
pico_enable_stdio_usb(bench_gpio_toggle 0)
pico_add_extra_outputs(bench_gpio_toggle)
