cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(bench_dma_memcpy C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(bench_dma_memcpy main.c)
target_compile_options(bench_dma_memcpy PRIVATE -O3 -flto)
target_link_options(bench_dma_memcpy PRIVATE -flto)
target_link_libraries(bench_dma_memcpy pico_stdlib hardware_dma hardware_clocks)
pico_enable_stdio_uart(bench_dma_memcpy 1)
pico_enable_stdio_usb(bench_dma_memcpy 0)
pico_add_extra_outputs(bench_dma_memcpy)
